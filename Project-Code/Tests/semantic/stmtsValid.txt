    a, b : integer;
    r    : real;
    s    : string;
    flag : boolean;

    begin
        w := v;
        return;
    end p;
    function f(n : value integer) return integer is
    begin
        return n * 2;
    end f;
begin
    a := 1;
    r := a;
    r := 2.5;
    s := "hi";
    flag := a < 10;
    p(a, b);
    b := f(3);
    if flag then null; end if;
    while b > 0 loop
        exit when b = 1;
        b := b - 1;
    end loop;
    for i in 1..5 loop
        b := b + i;
    end loop;
    read a;
    read(a, b);

end stmts;
